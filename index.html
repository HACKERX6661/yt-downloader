Optional Node.js Backend

For dedicated high-speed rendering on your own server.

This website already works on GitHub Pages using public API proxies. If you want your own server to process unlimited 4K 60FPS videos, you can host this Node.js snippet on Render or Koyeb:

1. Express Backend (`server.js`)
const express = require('express');
const { exec } = require('child_process');
const cors = require('cors');
const app = express();
app.use(cors());

app.get('/download', (req, res) => {
    const videoUrl = req.query.url;
    const cmd = `yt-dlp -f "bv*+ba/b" --merge-output-format mp4 -o - "${videoUrl}"`;
    res.header('Content-Disposition', 'attachment; filename="video.mp4"');
    res.header('Content-Type', 'video/mp4');
    exec(cmd).stdout.pipe(res);
});

app.listen(3000, () => console.log('Server active on port 3000'));
